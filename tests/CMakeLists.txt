cmake_minimum_required(VERSION 3.16)
project(IridiScopeIioTests CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_path(LIBIIO_INCLUDE_DIR NAMES iio.h REQUIRED)
enable_testing()
add_executable(iio_backend_test iio_backend_test.cpp
  ../src/sdr/iio_source.cpp ../src/sdr/live_iq_source.cpp)
target_include_directories(iio_backend_test PRIVATE ../src ${LIBIIO_INCLUDE_DIR})
target_compile_definitions(iio_backend_test PRIVATE HAS_LIBIIO=1 LIBIIO_STATIC=1)
target_link_libraries(iio_backend_test PRIVATE Threads::Threads)
add_test(NAME native_iio_backend COMMAND iio_backend_test)
set_tests_properties(native_iio_backend PROPERTIES TIMEOUT 15)
